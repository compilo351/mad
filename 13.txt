13.Write a program for spinner using event 
handler to display selected option name. 
XML::::::::::::::::::::::::::: 
<?xml version="1.0" encoding="utf-8"?> 
 
<LinearLayout 
xmlns:android="http://schemas.android.com/apk/re
s/android" 
    android:layout_width="match_parent" 
    android:layout_height="match_parent" 
    android:gravity="center" 
    android:orientation="vertical"> 
 
    <Spinner 
        android:id="@+id/spinner" 
        android:layout_width="wrap_content" 
        android:layout_height="wrap_content"/> 
 
</LinearLayout> 
JAVA::::::::::::::::: 
package com.example.app; 
 
import android.os.Bundle; 
import android.widget.*; 
import 
androidx.appcompat.app.AppCompatActivity; 
 
public class MainActivity extends 
AppCompatActivity { 
 
    protected void onCreate(Bundle b) { 
        super.onCreate(b); 
        setContentView(R.layout.activity_main); 
 
        Spinner sp = findViewById(R.id.sp); 
 
        String s[] = {"Java", "Python", "Android"}; 
 
        sp.setAdapter(new ArrayAdapter<>( 
                this, 
                
android.R.layout.simple_spinner_dropdown_item, 
                s)); 
 
        sp.setOnItemSelectedListener( 
                new 
AdapterView.OnItemSelectedListener() { 
 
                    public void onItemSelected( 
                            AdapterView<?> p, 
android.view.View v, 
                            int i, long l) { 
 
                        Toast.makeText( 
                                MainActivity.this, 
                                s[i], 
                                
Toast.LENGTH_SHORT).show(); 
                    } 
 
                    public void 
onNothingSelected(AdapterView<?> p) { 
                    } 
                }); 
    } 
}
